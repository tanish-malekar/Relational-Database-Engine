cmake_minimum_required(VERSION 3.22)
project(TransactionalSqlEngine VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(sqldb src/storage.cpp src/engine.cpp)
target_include_directories(sqldb PUBLIC include)
target_link_libraries(sqldb PUBLIC Threads::Threads)
target_compile_options(sqldb PRIVATE -Wall -Wextra -Wpedantic)
add_executable(sql_engine src/main.cpp)
target_link_libraries(sql_engine PRIVATE sqldb)
add_executable(sql_crash_worker tools/crash_worker.cpp)
target_link_libraries(sql_crash_worker PRIVATE sqldb)
include(CTest)
if(BUILD_TESTING)
 find_package(GTest CONFIG REQUIRED)
 add_executable(sql_tests tests/engine_test.cpp)
 target_link_libraries(sql_tests PRIVATE sqldb GTest::gtest_main)
 include(GoogleTest)
 gtest_discover_tests(sql_tests)
endif()
